import RPi.GPIO as GPIO
import time
import smtplib
from email.mime.text import MIMEText

# Configuration GPIO
RELAY_PIN = 26
GPIO.setmode(GPIO.BCM)
GPIO.setup(RELAY_PIN, GPIO.OUT)
GPIO.output(RELAY_PIN, GPIO.HIGH)  # Désactive le relai au démarrage

# Configuration email
EMAIL = "sifaouisamar03@gmail.com"
PASSWORD = "oemx bhaw lxqs rwxx"  # Mot de passe d'application
DESTINATAIRE = "sifaouisamar03@gmail.com"

def envoyer_email(sujet, message):
    try:
        msg = MIMEText(message)
        msg['Subject'] = sujet
        msg['From'] = EMAIL
        msg['To'] = DESTINATAIRE
        
        with smtplib.SMTP_SSL('smtp.gmail.com', 465) as server:
            server.login(EMAIL, PASSWORD)
            server.send_message(msg)
        print("Email envoyé : " + sujet)
    except Exception as e:
        print(f"Erreur envoi email: {e}")

try:
    while True:
        # Activation pompe (LOW pour activer)
        GPIO.output(RELAY_PIN, GPIO.LOW)
        print('Pompe ACTIVÉE - Arrosage en cours')
        envoyer_email("Arrosage démarré", "La pompe a été activée pour arrosage")
        
        time.sleep(10)  # Durée d'arrosage (10 secondes)
        
        # Désactivation pompe (HIGH pour désactiver)
        GPIO.output(RELAY_PIN, GPIO.HIGH)
        print('Pompe DÉSACTIVÉE - Arrosage terminé')
        envoyer_email("Arrosage terminé", "La pompe a été désactivée")
        
        time.sleep(10)  # Pause entre les arrosages (60 secondes)
        
finally:
    GPIO.output(RELAY_PIN, GPIO.HIGH)  # S'assure que la pompe est éteinte
    GPIO.cleanup()
