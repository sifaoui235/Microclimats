import sys
sys.path.insert(0, '/home/sifaoui/LCD')  # Adapter si besoin

import time
import board
import adafruit_dht
import drivers
import spidev
import RPi.GPIO as GPIO

# ------------------- Capteur DHT22 -------------------
dhtDevice = adafruit_dht.DHT22(board.D17)

# ------------------- LCD -------------------
display = drivers.Lcd()

# ------------------- Pompe & Capteur sol -------------------
POMPE_GPIO = 26
GPIO.setmode(GPIO.BCM)
GPIO.setwarnings(False)
GPIO.setup(POMPE_GPIO, GPIO.OUT, initial=GPIO.LOW)

DRY_SOL = 950
WET_SOL = 250
EAU_PAR_POURCENT = 4.2
EAU_0_A_80 = 80 * EAU_PAR_POURCENT

class SoilMoistureSensor:
    def __init__(self, channel=1):
        self.spi = spidev.SpiDev()
        self.spi.open(0, 0)
        self.spi.max_speed_hz = 1350000
        self.channel = channel

    def read_raw(self):
        adc_data = self.spi.xfer2([1, (8 + self.channel) << 4, 0])
        return ((adc_data[1] & 3) << 8) + adc_data[2]

    def read_percentage(self):
        raw = self.read_raw()
        return max(0, min(100, (DRY_SOL - raw) / (DRY_SOL - WET_SOL) * 100))

    def close(self):
        self.spi.close()

sensor = SoilMoistureSensor()
pompe_active = False

# ------------------- Réservoir (canal 1) -------------------
spi_res = spidev.SpiDev()
spi_res.open(0, 0)
spi_res.max_speed_hz = 1000000
VALEUR_MIN = 100
VALEUR_MAX = 950

def lire_canal(channel):
    adc = spi_res.xfer2([1, (8 + channel) << 4, 0])
    return ((adc[1] & 3) << 8) + adc[2]

# ------------------- Boucle principale -------------------
try:
    while True:
        # Lire humidité du sol
        moisture = sensor.read_percentage()

        # Contrôle de la pompe
        if moisture < 80:
            if not pompe_active:
                GPIO.output(POMPE_GPIO, GPIO.HIGH)
                pompe_active = True
                print("[Pompe] Activée (sol < 80%)")
        else:
            if pompe_active:
                GPIO.output(POMPE_GPIO, GPIO.LOW)
                pompe_active = False
                print("[Pompe] Désactivée (sol ≥ 80%)")

        # ----------- Température & Humidité ---------
        try:
            temperature_c = dhtDevice.temperature
            humidity = dhtDevice.humidity

            if temperature_c is not None and humidity is not None:
                display.lcd_clear()
                display.lcd_display_string(f"Temp: {temperature_c:.1f}C", 1)
                display.lcd_display_string(f"Humidity: {humidity:.1f}%", 2)
                time.sleep(4)
            else:
                print("Lecture DHT invalide")
                time.sleep(2)

        except RuntimeError as error:
            print("Erreur DHT:", error.args[0])
            time.sleep(2)

        # ----------- Humidité du sol + affichage ---------
        display.lcd_clear()
        display.lcd_display_string(f"0→80%: {EAU_0_A_80:.1f}mL", 1)
        display.lcd_display_string(f"Actuel: {moisture:.1f}%", 2)
        print(f"[Sol] Humidité: {moisture:.1f}%")
        time.sleep(4)

        # ----------- Réservoir ---------
        valeur_brute = lire_canal(1)
        pourcentage = (valeur_brute - VALEUR_MIN) / (VALEUR_MAX - VALEUR_MIN) * 100
        pourcentage = max(0, min(100, round(pourcentage, 1)))

        print(f"[Réservoir] Niveau: {pourcentage}% | Raw: {valeur_brute}")
        display.lcd_clear()
        display.lcd_display_string(f"Niveau: {pourcentage}%", 1)
        display.lcd_display_string(f"Raw: {valeur_brute}", 2)
        time.sleep(4)

except KeyboardInterrupt:
    print("Fin du programme")
    GPIO.output(POMPE_GPIO, GPIO.LOW)
    sensor.close()
    spi_res.close()
    display.lcd_clear()
    GPIO.cleanup()
